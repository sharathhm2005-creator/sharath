from flask import Flask, render_template, request

app = Flask(__name__)

@app.route('/')
def home():
    profile = {
        "name": "Sharath H M",
        "role": "Python Developer",
        "about": "I am a passionate Python developer interested in Web Development, Data Science, and AI."
    }
    return render_template("index.html", profile=profile)

@app.route('/contact', methods=['POST'])
def contact():
    name = request.form['name']
    email = request.form['email']
    message = request.form['message']

    return f"""
    <h2>Thank You!</h2>
    <p>Name: {name}</p>
    <p>Email: {email}</p>
    <p>Message: {message}</p>
    <a href="/">Go Back</a>
    """

if __name__ == '__main__':
    app.run(debug=True)
